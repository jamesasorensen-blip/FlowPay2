function guestFormatDate(value){
 const match=/^(\d{4})-(\d{2})-(\d{2})$/.exec(String(value||''));
 return match?match[2]+'/'+match[3]+'/'+match[1]:String(value||'');
}
function guestParseDate(value){
 const m=/^(\d{2})\/(\d{2})\/(\d{4})$/.exec(String(value||''));if(!m)return '';
 const d=new Date(Number(m[3]),Number(m[1])-1,Number(m[2]));
 return d.getFullYear()===Number(m[3])&&d.getMonth()===Number(m[1])-1&&d.getDate()===Number(m[2])?m[3]+'-'+m[1]+'-'+m[2]:'';
}
function guestFormatTimestamp(value){
 const d=new Date(value);if(!Number.isFinite(d.getTime()))return '';
 const day=[d.getFullYear(),String(d.getMonth()+1).padStart(2,'0'),String(d.getDate()).padStart(2,'0')].join('-');
 return guestFormatDate(day)+' '+d.toLocaleTimeString();
}
function initGuestDateInputs(){
 // Keep application-facing values ISO; present strict MM/DD/YYYY in every field.
 const native=Object.getOwnPropertyDescriptor(HTMLInputElement.prototype,'value');
 for(const input of document.querySelectorAll('input[type="date"]')){
  const original=native.get.call(input);input.type='text';input.placeholder='MM/DD/YYYY';input.inputMode='numeric';input.maxLength=10;input.pattern='[0-9]{2}/[0-9]{2}/[0-9]{4}';
  input.setAttribute('aria-description','Date in MM/DD/YYYY format');
  function validate(){const v=native.get.call(input);input.setCustomValidity(v&&!guestParseDate(v)?'Enter a valid date in MM/DD/YYYY format.':'')}
  Object.defineProperty(input,'value',{configurable:true,get(){return guestParseDate(native.get.call(input))},set(value){native.set.call(input,guestFormatDate(value));validate()}});
  input.value=original;input.addEventListener('input',validate);input.addEventListener('change',validate);
  input.form?.addEventListener('reset',()=>queueMicrotask(validate));
 }
}
document.addEventListener?.('DOMContentLoaded',()=>queueMicrotask(initGuestDateInputs),{once:true});
