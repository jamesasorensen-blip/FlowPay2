function guestBillIcon(b){
 const chosen=guestBillIcons.find(x=>x[0]===b.icon&&x[0]!=='auto');if(chosen)return chosen;
 const text=(b.category+' '+b.name).toLowerCase();
 const rules=[['electric',/electric|power|energy/],['internet',/internet|broadband|wifi/],['phone',/phone|mobile|cellular/],['prescription',/prescription|pharmacy|medication/],['health',/health insurance/],['insurance',/insurance/],['home',/rent|mortgage|housing/],['car',/car payment|auto loan|vehicle/],['student',/student|tuition|education/],['medical',/medical|hospital|doctor/],['credit',/credit card/],['loan',/loan/],['gym',/gym|fitness/],['stream',/streaming|netflix|hulu|disney|subscription/],['water',/water|sewer/],['gas',/natural gas|gas bill/],['food',/grocery|groceries/],['pet',/pet|veterinary/]];
 const id=rules.find(x=>x[1].test(text))?.[0]||'bill';return guestBillIcons.find(x=>x[0]===id);
}
